// =====================================================
// CẤU HÌNH CẢNH BÁO QUẢN TRỊ
// Có thể điều chỉnh các ngưỡng này sau này nếu cần.
// =====================================================
const KPI_WARNING = {
    TOP_REVENUE_COUNT: 2,          // Chỉ xét Top 2 doanh thu
    BILL_AVG_DROP_THRESHOLD: 0.10, // Bill TB thấp >= 10% so với TB salon
    HC_AVG_DROP_THRESHOLD: 0.10,   // Bill HC TB thấp >= 10% so với TB salon
    SEVERE_DROP_THRESHOLD: 0.20    // Thấp >= 20% -> Cần chú ý
};

function getLastNonEmptyRowInRange(sheet, col, startRow, endRow) {
    var values = sheet.getRange(startRow, col, endRow - startRow + 1, 1).getValues();

    for (var i = values.length - 1; i >= 0; i--) {
        if (values[i][0] !== "" && values[i][0] !== null) {
            return startRow + i;
        }
    }

    return startRow - 1;
}

function breakAllMergedRanges(range) {
    var mergedRanges = range.getMergedRanges();

    for (var i = 0; i < mergedRanges.length; i++) {
        mergedRanges[i].breakApart();
    }
}

function normalizeTenBill(value) {
    return String(value || "").trim().toUpperCase();
}

function isBillKYC(value) {
    return normalizeTenBill(value).includes("KYC");
}

function isBillBSP(value) {
    return normalizeTenBill(value) === "BSP";
}

function isBillHoaChat(value) {
    var tenBill = normalizeTenBill(value);

    if (!tenBill) {
        return false;
    }

    // Không tính các bill này là hóa chất
    var exclude = [
        "C",
        "CẮT",
        "CAT",
        "G",
        "GỘI",
        "GOI",
        "DVL",
        "BSP"
    ];

    return !exclude.includes(tenBill);
}

function clearThongKeNhanVienCu(sheet) {
    var rangeThongKe = sheet.getRange(
        COL.KPI.START_ROW,
        COL.KPI.START_COL,
        50,
        COL.KPI.NUM_COLS
    );

    breakAllMergedRanges(rangeThongKe);

    rangeThongKe
        .clearContent()
        .clearFormat()
        .setBackground("#ffffff")
        .setBorder(false, false, false, false, false, false)
        .setFontFamily("Arial")
        .setFontSize(13)
        .setFontWeight("normal");
}

function highlightSpecialBills(
    sheet,
    startRow,
    customerValues
) {

    var COL_CUSTOMER = 15; // O
    var COL_BILL = 16;     // P

    for (
        var i = 0;
        i < customerValues.length;
        i++
    ) {

        var value = String(
            customerValues[i][0] || ""
        )
            .trim()
            .toUpperCase();

        // Range gồm O:P của cùng 1 dòng
        var highlightRange = sheet.getRange(
            startRow + i,
            COL_CUSTOMER,
            1,
            2
        );

        // =========================
        // KYC -> VÀNG
        // =========================
        if (value.includes("KYC")) {

            highlightRange
                .setBackground("#ffff00")
                .setFontColor("#000000");

        // =========================
        // BSP -> XANH DƯƠNG
        // =========================
        } else if (value === "BSP") {

            highlightRange
                .setBackground("#00ffff")
                .setFontColor("#000000");
        }
    }
}


// =====================================================
// ĐÁNH GIÁ CẢNH BÁO QUẢN TRỊ CHO THỢ CHÍNH
//
// Rule hiện tại:
// - Chỉ xét Top 2 doanh thu.
// - Bill TB thấp >= 10% so với Bill TB salon => cảnh báo.
// - Nếu Bill HC TB cũng thấp >= 10% => nâng mức chú ý.
// - Nếu Bill TB thấp >= 20% => mức "Cần chú ý".
//
// Hàm trả về:
// - billTrungBinhSalon
// - billHoaChatTrungBinhSalon
// - warnings: danh sách cảnh báo để hiển thị tại I:L
// - statusByName: trạng thái để hiển thị ở bảng tổng quát N:AC
// =====================================================
function taoDanhGiaQuanTri(listThoChinh) {
    var tongDoanhThuThoChinh = 0;
    var tongBillThoChinh = 0;
    var tongDoanhThuHoaChat = 0;
    var tongBillHoaChat = 0;

    listThoChinh.forEach(function (tho) {
        tongDoanhThuThoChinh += Number(tho.tongDoanhThuCaNhan) || 0;
        tongBillThoChinh += Number(tho.tongBill) || 0;
        tongDoanhThuHoaChat += Number(tho.tongDoanhThuHoaChat) || 0;
        tongBillHoaChat += Number(tho.soBillHoaChat) || 0;
    });

    var billTrungBinhSalon = tongBillThoChinh > 0
        ? tongDoanhThuThoChinh / tongBillThoChinh
        : 0;

    var billHoaChatTrungBinhSalon = tongBillHoaChat > 0
        ? tongDoanhThuHoaChat / tongBillHoaChat
        : 0;

    // Mặc định tất cả nhân viên là bình thường
    var statusByName = {};

    listThoChinh.forEach(function (tho) {
        statusByName[tho.name] = {
            text: "✓ Bình thường",
            level: "NORMAL",
            background: "#d9ead3",
            chenhLech: 0,
            rankDoanhThu: 0
        };
    });

    // Chỉ đưa những người thực sự có bill vào bảng xếp hạng doanh thu
    var xepHangDoanhThu = listThoChinh
        .map(function (tho, index) {
            return {
                tho: tho,
                originalIndex: index
            };
        })
        .filter(function (item) {
            return (Number(item.tho.tongDoanhThuCaNhan) || 0) > 0 &&
                (Number(item.tho.tongBill) || 0) > 0;
        })
        .sort(function (a, b) {
            var diff =
                (Number(b.tho.tongDoanhThuCaNhan) || 0) -
                (Number(a.tho.tongDoanhThuCaNhan) || 0);

            // Nếu doanh thu bằng nhau thì giữ thứ tự cấu hình
            return diff !== 0 ? diff : a.originalIndex - b.originalIndex;
        });

    var warnings = [];
    var soNguoiCanXet = Math.min(
        KPI_WARNING.TOP_REVENUE_COUNT,
        xepHangDoanhThu.length
    );

    for (var i = 0; i < soNguoiCanXet; i++) {
        var tho = xepHangDoanhThu[i].tho;
        var doanhThu = Number(tho.tongDoanhThuCaNhan) || 0;
        var tongBill = Number(tho.tongBill) || 0;

        var billTrungBinh = tongBill > 0
            ? doanhThu / tongBill
            : 0;

        if (billTrungBinhSalon <= 0 || billTrungBinh <= 0) {
            continue;
        }

        // Ví dụ thấp hơn salon 18.5% => 0.185
        var tyLeThapHon =
            (billTrungBinhSalon - billTrungBinh) /
            billTrungBinhSalon;

        if (tyLeThapHon < KPI_WARNING.BILL_AVG_DROP_THRESHOLD) {
            continue;
        }

        var billHoaChatTrungBinh = 0;

        if ((Number(tho.soBillHoaChat) || 0) > 0) {
            billHoaChatTrungBinh =
                (Number(tho.tongDoanhThuHoaChat) || 0) /
                (Number(tho.soBillHoaChat) || 0);
        }

        var tyLeHoaChatThapHon = 0;

        if (
            billHoaChatTrungBinhSalon > 0 &&
            billHoaChatTrungBinh > 0 &&
            billHoaChatTrungBinh < billHoaChatTrungBinhSalon
        ) {
            tyLeHoaChatThapHon =
                (billHoaChatTrungBinhSalon - billHoaChatTrungBinh) /
                billHoaChatTrungBinhSalon;
        }

        var hoaChatCungThap =
            tyLeHoaChatThapHon >= KPI_WARNING.HC_AVG_DROP_THRESHOLD;

        var canChuY =
            tyLeThapHon >= KPI_WARNING.SEVERE_DROP_THRESHOLD ||
            hoaChatCungThap;

        var canhBaoText = hoaChatCungThap
            ? "DT Top #" + (i + 1) + " - Bill TB & HC TB thấp"
            : "DT Top #" + (i + 1) + " - Bill TB thấp";

        var mucText = canChuY
            ? "🔴 Cần chú ý"
            : "⚠ Cảnh báo";

        var statusText = hoaChatCungThap
            ? (canChuY ? "🔴 Bill TB & HC TB thấp" : "⚠ Bill TB & HC TB thấp")
            : (canChuY ? "🔴 Bill TB thấp" : "⚠ Bill TB thấp");

        var background = canChuY
            ? "#f4cccc"
            : "#fce5cd";

        var warning = {
            name: tho.name,
            color: tho.color,
            canhBao: canhBaoText,
            // Giá trị âm để trên sheet hiển thị -18.5%
            chenhLech: -tyLeThapHon,
            muc: mucText,
            level: canChuY ? "DANGER" : "WARNING",
            background: background,
            rankDoanhThu: i + 1,
            billTrungBinh: billTrungBinh,
            billTrungBinhSalon: billTrungBinhSalon,
            billHoaChatTrungBinh: billHoaChatTrungBinh,
            billHoaChatTrungBinhSalon: billHoaChatTrungBinhSalon
        };

        warnings.push(warning);

        statusByName[tho.name] = {
            text: statusText,
            level: warning.level,
            background: background,
            chenhLech: warning.chenhLech,
            rankDoanhThu: warning.rankDoanhThu
        };
    }

    return {
        billTrungBinhSalon: billTrungBinhSalon,
        billHoaChatTrungBinhSalon: billHoaChatTrungBinhSalon,
        warnings: warnings,
        statusByName: statusByName
    };
}

function xuatThongKeNhanVien(sheet, startRow, listThoChinh, listThoPhu) {
    var startCol = COL.KPI.START_COL; // I
    var totalCols = COL.KPI.NUM_COLS; // I:L
    var fontSize = 13;

    sheet.setColumnWidth(9, 115); // I
    sheet.setColumnWidth(10, 125); // J
    sheet.setColumnWidth(11, 135); // K
    sheet.setColumnWidth(12, 120); // L

    // =====================================================
    // TITLE
    // =====================================================
    sheet.getRange(startRow, startCol, 1, totalCols)
        .merge()
        .setValue("THỐNG KÊ KPI NHÂN VIÊN")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d9ead3")
        .setBorder(true, true, true, true, true, true);

    // =====================================================
    // 1. TỔNG QUAN
    // =====================================================
    var rowTongQuanTitle = startRow + 1;
    var rowTongQuanHeader = startRow + 2;
    var rowTongQuanData = startRow + 3;

    sheet.getRange(rowTongQuanTitle, startCol, 1, totalCols)
        .merge()
        .setValue("TỔNG QUAN")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")
        .setBackground("#eeeeee")
        .setBorder(true, true, true, true, true, true);

    sheet.getRange(rowTongQuanHeader, startCol, 1, totalCols)
        .setValues([[
            "Nhân viên",
            "Tổng lương",
            "DT riêng",
            "Lương / DT"
        ]])
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true);

    var tongQuanRows = [];

    listThoChinh.forEach(function (tho) {
        var tyLeLuong = tho.tongDoanhThuCaNhan > 0
            ? tho.tongLuong / tho.tongDoanhThuCaNhan
            : 0;

        tongQuanRows.push([
            tho.name,
            tho.tongLuong,
            tho.tongDoanhThuCaNhan,
            tyLeLuong
        ]);
    });

    listThoPhu.forEach(function (tho) {
        var tyLeLuong = tho.tongDoanhThuCaNhan > 0
            ? tho.tongLuong / tho.tongDoanhThuCaNhan
            : 0;

        tongQuanRows.push([
            tho.name,
            tho.tongLuong,
            tho.tongDoanhThuCaNhan,
            tyLeLuong
        ]);
    });

    if (tongQuanRows.length > 0) {
        sheet.getRange(rowTongQuanData, startCol, tongQuanRows.length, totalCols)
            .setValues(tongQuanRows)
            .setFontSize(fontSize)
            .setFontWeight("normal")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);

        sheet.getRange(rowTongQuanData, startCol + 1, tongQuanRows.length, 2)
            .setNumberFormat("#,##0");

        sheet.getRange(rowTongQuanData, startCol + 3, tongQuanRows.length, 1)
            .setNumberFormat("0.00%");
    }

    for (var i = 0; i < listThoChinh.length; i++) {
        sheet.getRange(rowTongQuanData + i, startCol)
            .setBackground(listThoChinh[i].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }

    for (var j = 0; j < listThoPhu.length; j++) {
        sheet.getRange(rowTongQuanData + listThoChinh.length + j, startCol)
            .setBackground(listThoPhu[j].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }

    // =====================================================
    // 2. DOANH THU ĐẶC BIỆT
    // KYC chỉ có thợ chính, BSP có cả thợ chính và thợ phụ
    // =====================================================
    var rowDoanhThuTitle = rowTongQuanData + tongQuanRows.length + 1;
    var rowDoanhThuHeader = rowDoanhThuTitle + 1;
    var rowDoanhThuData = rowDoanhThuHeader + 1;

    sheet.getRange(rowDoanhThuTitle, startCol, 1, totalCols)
        .merge()
        .setValue("DOANH THU ĐẶC BIỆT")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")
        .setBackground("#eeeeee")
        .setBorder(true, true, true, true, true, true);

    sheet.getRange(rowDoanhThuHeader, startCol, 1, totalCols)
        .setValues([[
            "Nhân viên",
            "Vai trò",
            "DT KYC",
            "DT BSP"
        ]])
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true);

    var doanhThuRows = [];

    listThoChinh.forEach(function (tho) {
        doanhThuRows.push([
            tho.name,
            "Thợ chính",
            tho.tongDoanhThuKYC,
            tho.tongDoanhThuBSP

    });

    listThoPhu.forEach(function (tho) {
        doanhThuRows.push([
            tho.name,
            "Thợ phụ",

            tho.tongDoanhThuBSP

    });

        sheet.getRange(rowDoanhThuData, startCol, doanhThuRows.length, totalCols)
            .setValues(doanhThuRows)
            .setFontSize(fontSize)
            .setFontWeight("normal")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);

        sheet.getRange(rowDoanhThuData, startCol + 2, doanhThuRows.length, 2)
            .setNumberFormat("#,##0");
    }

    for (var k = 0; k < listThoChinh.length; k++) {
        sheet.getRange(rowDoanhThuData + k, startCol)
            .setBackground(listThoChinh[k].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }

    for (var p = 0; p < listThoPhu.length; p++) {
        sheet.getRange(rowDoanhThuData + listThoChinh.length + p, startCol)
            .setBackground(listThoPhu[p].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }

    // =====================================================
    // 3. HÓA CHẤT - THỢ CHÍNH
    // =====================================================
    var rowHoaChatTitle = rowDoanhThuData + doanhThuRows.length + 1;
    var rowHoaChatHeader = rowHoaChatTitle + 1;
    var rowHoaChatData = rowHoaChatHeader + 1;

    sheet.getRange(rowHoaChatTitle, startCol, 1, totalCols)
        .merge()
        .setValue("HÓA CHẤT - THỢ CHÍNH")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")
        .setBackground("#eeeeee")
        .setBorder(true, true, true, true, true, true);

    sheet.getRange(rowHoaChatHeader, startCol, 1, totalCols)

            "Nhân viên",
            "SL bill HC",
            "DT hóa chất",
            "Bill TB"

        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true);

    var hoaChatRows = [];

    listThoChinh.forEach(function (tho) {
        var billTrungBinh = tho.soBillHoaChat > 0
            ? tho.tongDoanhThuHoaChat / tho.soBillHoaChat
            : 0;

        hoaChatRows.push([
            tho.name,
            tho.soBillHoaChat,
            tho.tongDoanhThuHoaChat,
            billTrungBinh
        ]);
    });

    if (hoaChatRows.length > 0) {
        sheet.getRange(rowHoaChatData, startCol, hoaChatRows.length, totalCols)
            .setValues(hoaChatRows)
            .setFontSize(fontSize)
            .setFontWeight("normal")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);

        sheet.getRange(rowHoaChatData, startCol + 1, hoaChatRows.length, 1)
            .setNumberFormat("0");

        sheet.getRange(rowHoaChatData, startCol + 2, hoaChatRows.length, 2)
            .setNumberFormat("#,##0");
    }

    for (var h = 0; h < listThoChinh.length; h++) {
        sheet.getRange(rowHoaChatData + h, startCol)
            .setBackground(listThoChinh[h].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)

    // =====================================================
    // 4. CẢNH BÁO QUẢN TRỊ
    // Hiển thị sau bảng HÓA CHẤT - THỢ CHÍNH, cách 1 dòng.
    // =====================================================

        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")

        .setBorder(true, true, true, true, true, true);

    if (canhBaoRows.length === 0) {
        // Không có cảnh báo: chỉ hiển thị 1 dòng trạng thái

            .merge()
            .setValue("✓ Không có cảnh báo quản trị đáng chú ý")
            .setFontSize(fontSize)
            .setFontWeight("bold")
            .setFontColor("#274e13")
            .setBackground("#d9ead3")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);
    } else {
        var rowCanhBaoHeader = rowCanhBaoTitle + 1;
        var rowCanhBaoData = rowCanhBaoHeader + 1;

        sheet.getRange(rowCanhBaoHeader, startCol, 1, totalCols)
            .setValues([[
                "Nhân viên",
                "Cảnh báo",
                "So với salon",
                "Mức"
            ]])
            .setFontSize(fontSize)
            .setFontWeight("bold")
            .setBackground("#d3d3d3")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setWrap(true)
            .setBorder(true, true, true, true, true, true);

        var warningValues = canhBaoRows.map(function (item) {
            return [
                item.name,
                item.canhBao,
                item.chenhLech,
                item.muc
            ];
        });

        sheet.getRange(
            rowCanhBaoData,
            startCol,
            warningValues.length,
            totalCols
        )
            .setValues(warningValues)
            .setFontSize(fontSize)
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setWrap(true)
            .setBorder(true, true, true, true, true, true);

        // Chênh lệch so với Bill TB salon
        sheet.getRange(
            rowCanhBaoData,
            startCol + 2,
            warningValues.length,
            1
        )
            .setNumberFormat("0.0%")
            .setFontWeight("bold");

        for (var cb = 0; cb < canhBaoRows.length; cb++) {
            // Tên nhân viên: giữ màu nhân viên
            sheet.getRange(rowCanhBaoData + cb, startCol)
                .setBackground(canhBaoRows[cb].color)
                .setFontColor("#000000")
                .setFontWeight("bold")
                .setHorizontalAlignment("left");

            // Ô mức cảnh báo
            sheet.getRange(rowCanhBaoData + cb, startCol + 3)
                .setBackground(canhBaoRows[cb].background)
                .setFontWeight("bold");

            // Ô chênh lệch
            sheet.getRange(rowCanhBaoData + cb, startCol + 2)
                .setBackground(canhBaoRows[cb].background);
        }
    }

}


// =====================================================
// BẢNG TỔNG QUÁT HIỆU SUẤT THỢ CHÍNH
// Bắt đầu từ cột N và được đặt dưới bảng lương 1 hàng trống.
// =====================================================
function xuatBangTongQuatThoChinh(
    sheet,
    startRow,
    listThoChinh,
    tongDoanhThuSalon
) {
    var startCol = COL.SALARY.START; // N = 14
    var fontSize = 13;

    // Dùng cùng 1 rule với khối CẢNH BÁO QUẢN TRỊ ở I:L
    var danhGiaQuanTri = taoDanhGiaQuanTri(listThoChinh);

    var headers = [
        "Thợ",
        "Doanh thu",
        "% DT salon",
        "Tổng bill",
        "Bill TB",
        "SL bill HC",
        "DT hóa chất",
        "Bill HC TB",
        "SL KYC",
        "DT KYC",
        "Ngày làm",
        "DT / ngày",
        "Bill / ngày",
        "Tổng lương",
        "Lương / DT",
        "Trạng thái"
    ];

    var totalCols = headers.length;

    // =================================================
    // TITLE
    // =================================================
    sheet.getRange(startRow, startCol, 1, totalCols)
        .merge()
        .setValue("TỔNG QUÁT HIỆU SUẤT THỢ CHÍNH")
        .setFontFamily("Arial")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d9ead3")
        .setBorder(true, true, true, true, true, true);

    // =================================================
    // HEADER
    // =================================================
    sheet.getRange(startRow + 1, startCol, 1, totalCols)
        .setValues([headers])
        .setFontFamily("Arial")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setBackground("#d3d3d3")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setWrap(true)
        .setBorder(true, true, true, true, true, true);

    if (listThoChinh.length === 0) {
        return;
    }

    // =================================================
    // BUILD DATA
    // =================================================
    var rows = [];

    listThoChinh.forEach(function (tho) {
        var tongBill = Number(tho.tongBill) || 0;
        var soBillHoaChat = Number(tho.soBillHoaChat) || 0;
        var soBillKYC = Number(tho.soBillKYC) || 0;
        var doanhThu = Number(tho.tongDoanhThuCaNhan) || 0;
        var doanhThuHoaChat = Number(tho.tongDoanhThuHoaChat) || 0;
        var doanhThuKYC = Number(tho.tongDoanhThuKYC) || 0;
        var tongLuong = Number(tho.tongLuong) || 0;

        var soNgayLam = Object.keys(tho.ngayLamMap || {}).length;

        var tyTrongDoanhThu = tongDoanhThuSalon > 0
            ? doanhThu / tongDoanhThuSalon
            : 0;

        var billTrungBinh = tongBill > 0
            ? doanhThu / tongBill
            : 0;

        var billHoaChatTrungBinh = soBillHoaChat > 0
            ? doanhThuHoaChat / soBillHoaChat
            : 0;

        var doanhThuNgay = soNgayLam > 0
            ? doanhThu / soNgayLam
            : 0;

        var billNgay = soNgayLam > 0
            ? tongBill / soNgayLam
            : 0;

        var tyLeLuong = doanhThu > 0
            ? tongLuong / doanhThu
            : 0;

        var trangThai = danhGiaQuanTri.statusByName[tho.name]
            ? danhGiaQuanTri.statusByName[tho.name].text
            : "✓ Bình thường";

        rows.push([
            tho.name,
            doanhThu,
            tyTrongDoanhThu,
            tongBill,
            billTrungBinh,
            soBillHoaChat,
            doanhThuHoaChat,
            billHoaChatTrungBinh,
            soBillKYC,
            doanhThuKYC,
            soNgayLam,
            doanhThuNgay,
            billNgay,
            tongLuong,
            tyLeLuong,
            trangThai
        ]);
    });

    var dataStartRow = startRow + 2;

    // =================================================
    // WRITE DATA
    // =================================================
    sheet.getRange(
        dataStartRow,
        startCol,
        rows.length,
        totalCols
    )
        .setValues(rows)
        .setFontFamily("Arial")
        .setFontSize(fontSize)
        .setFontWeight("normal")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBorder(true, true, true, true, true, true);

    // Tên thợ: màu theo cấu hình
    for (var i = 0; i < listThoChinh.length; i++) {
        sheet.getRange(dataStartRow + i, startCol)
            .setBackground(listThoChinh[i].color)
            .setFontColor("#000000")
            .setFontWeight("bold");

        var statusInfo = danhGiaQuanTri.statusByName[listThoChinh[i].name] || {
            background: "#d9ead3"
        };

        // Cột AC = Trạng thái
        sheet.getRange(dataStartRow + i, startCol + 15)
            .setBackground(statusInfo.background)
            .setFontWeight("bold")
            .setWrap(true);
    }

    // =================================================
    // FORMAT
    // =================================================

    // Doanh thu
    sheet.getRange(dataStartRow, startCol + 1, rows.length, 1)
        .setNumberFormat("#,##0");

    // % DT salon
    sheet.getRange(dataStartRow, startCol + 2, rows.length, 1)
        .setNumberFormat("0.00%");

    // Tổng bill
    sheet.getRange(dataStartRow, startCol + 3, rows.length, 1)
        .setNumberFormat("0");

    // Bill TB
    sheet.getRange(dataStartRow, startCol + 4, rows.length, 1)
        .setNumberFormat("#,##0");

    // SL bill HC
    sheet.getRange(dataStartRow, startCol + 5, rows.length, 1)
        .setNumberFormat("0");

    // DT hóa chất + Bill HC TB
    sheet.getRange(dataStartRow, startCol + 6, rows.length, 2)
        .setNumberFormat("#,##0");

    // SL KYC
    sheet.getRange(dataStartRow, startCol + 8, rows.length, 1)
        .setNumberFormat("0");

    // DT KYC
    sheet.getRange(dataStartRow, startCol + 9, rows.length, 1)
        .setNumberFormat("#,##0");

    // Ngày làm
    sheet.getRange(dataStartRow, startCol + 10, rows.length, 1)
        .setNumberFormat("0");

    // DT / ngày
    sheet.getRange(dataStartRow, startCol + 11, rows.length, 1)
        .setNumberFormat("#,##0");

    // Bill / ngày
    sheet.getRange(dataStartRow, startCol + 12, rows.length, 1)
        .setNumberFormat("0.00");

    // Tổng lương
    sheet.getRange(dataStartRow, startCol + 13, rows.length, 1)
        .setNumberFormat("#,##0");

    // Lương / DT
    sheet.getRange(dataStartRow, startCol + 14, rows.length, 1)
        .setNumberFormat("0.00%");

    // =================================================
    // WIDTH
    // =================================================
    var widths = [
        115, // Thợ
        130, // Doanh thu
        105, // % DT salon
        95,  // Tổng bill
        115, // Bill TB
        100, // SL bill HC
        130, // DT hóa chất
        120, // Bill HC TB
        90,  // SL KYC
        115, // DT KYC
        90,  // Ngày làm
        120, // DT / ngày
        100, // Bill / ngày
        125, // Tổng lương
        105, // Lương / DT
        190  // Trạng thái
    ];

    for (var c = 0; c < widths.length; c++) {
        sheet.setColumnWidth(startCol + c, widths[c]);
    }
}

function createOrUpdateSheetLuong() {

        return;
    }

    var lastColumn = sheet.getLastColumn();

    if (lastColumn >= COL.SALARY.START) {
        var numCols = lastColumn - COL.SALARY.START + 1;
        var rangeUnMerge = sheet.getRange(3, COL.SALARY.START, 2, numCols);

            .clearContent()

    sheet.setColumnWidth(14, 120)
        .setColumnWidth(15, 150)
        .setColumnWidth(16, 130)
        .setColumnWidth(17, 130)
        .setColumnWidth(18, 150)
        .setColumnWidth(19, 150)
        .setColumnWidth(20, 150)
        .setColumnWidth(21, 150)
        .setColumnWidth(22, 150)
        .setColumnWidth(23, 150)
        .setColumnWidth(24, 150)
        .setColumnWidth(25, 150);

    const cellTitle = sheet.getRange("N1");
    let title = "Lương " + sheet.getRange("A1").getValue();

    cellTitle
        .setValue(title)
        .setFontWeight("bold")
        .setFontSize(17)
        .setHorizontalAlignment("left");

    var values = [["Ngày", "Tên khách", "Tiền bill", "Tổng bill ngày"]];

    const headerCommons = sheet.getRange(3, COL.SALARY.DATE, 2, 4);

    for (var col = COL.SALARY.DATE; col <= COL.SALARY.DAY_TOTAL; col++) {
        sheet.getRange(3, col, 2, 1).merge();
        sheet.getRange(3, col).setValue(values[0][col - COL.SALARY.DATE]);
        sheet.getRange(3, col)
            .setVerticalAlignment("middle")
            .setHorizontalAlignment("center");
    }

    headerCommons
        .setFontWeight("bold")
        .setFontSize(12)
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true);

    var lastRowThoChinh = getLastNonEmptyRowInRange(
        sheet,
        COL.CONFIG.MAIN_NAME,
        COL.CONFIG.START_ROW,
        COL.CONFIG.END_ROW
    );

    var lastRowThoPhu = getLastNonEmptyRowInRange(
        sheet,
        COL.CONFIG.ASSISTANT_NAME,
        COL.CONFIG.START_ROW,
        COL.CONFIG.END_ROW
    );

    var thoChinh = [];
    var thoPhu = [];
    var colThoChinh = null;
    var colThoPhu = null;

    if (lastRowThoChinh >= COL.CONFIG.START_ROW) {
        colThoChinh = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.MAIN_NAME,
            lastRowThoChinh - COL.CONFIG.START_ROW + 1,
            1
        );
        thoChinh = colThoChinh.getValues().filter(function (r) { return r[0] !== ""; });
    }

    if (lastRowThoPhu >= COL.CONFIG.START_ROW) {
        colThoPhu = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.ASSISTANT_NAME,
            lastRowThoPhu - COL.CONFIG.START_ROW + 1,
            1
        );
        thoPhu = colThoPhu.getValues().filter(function (r) { return r[0] !== ""; });
    }

    if (thoChinh.length === 0 && thoPhu.length === 0) {
        ui.alert("Chưa có danh sách nhân viên trong vùng cấu hình I4:L12.");
        return;
    }

    var startCellTho = sheet.getRange(4, COL.SALARY.WORKER_START); // R4
    var countTho = 0;

    for (var i = 0; i < thoChinh.length; i++) {
        var sourceCellMain = sheet.getRange(COL.CONFIG.START_ROW + i, COL.CONFIG.MAIN_NAME);

        startCellTho.offset(0, countTho)
            .setValue(thoChinh[i][0])
            .setFontWeight("bold")
            .setFontSize(12)
            .setBackground(sourceCellMain.getBackground())
            .setBorder(true, true, true, true, true, true)
            .setHorizontalAlignment("center");

        countTho++;
    }

    if (thoChinh.length > 0) {
        sheet.getRange(3, COL.SALARY.WORKER_START, 1, thoChinh.length).merge();
        sheet.getRange(3, COL.SALARY.WORKER_START)
            .setValue("Thợ chính")
            .setFontWeight("bold")
            .setFontSize(12)
            .setBackground("#d3d3d3")
            .setBorder(true, true, true, true, true, true)
            .setHorizontalAlignment("center");
    }

    for (var j = 0; j < thoPhu.length; j++) {
        var sourceCellAssistant = sheet.getRange(COL.CONFIG.START_ROW + j, COL.CONFIG.ASSISTANT_NAME);

        startCellTho.offset(0, countTho)
            .setValue(thoPhu[j][0])
            .setFontWeight("bold")
            .setFontSize(12)
            .setBackground(sourceCellAssistant.getBackground())
            .setBorder(true, true, true, true, true, true)
            .setHorizontalAlignment("center");

        countTho++;
    }

    if (thoPhu.length > 0) {
        sheet.getRange(
            3,
            COL.SALARY.WORKER_START + thoChinh.length,
            1,
            thoPhu.length
        ).merge();

        sheet.getRange(3, COL.SALARY.WORKER_START + thoChinh.length)
            .setValue("Thợ phụ")
            .setFontWeight("bold")
            .setFontSize(12)
            .setBackground("#d3d3d3")
            .setBorder(true, true, true, true, true, true)
            .setHorizontalAlignment("center");
    }

    sheet.getRange(
        3,
        COL.SALARY.WORKER_START + thoChinh.length + thoPhu.length,
        2,
        1
    )
        .merge()
        .setValue("Ngày sửa")
        .setFontWeight("bold")
        .setFontSize(12)
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true)
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");
}

function tinhLuong() {
    var ui = SpreadsheetApp.getUi();

    // =====================================================
    // CẤU HÌNH VỊ TRÍ
    // =====================================================
    var dongTongLuong = 5;      // Tổng nằm dòng 5-6
    var dongBatDauLuong = 7;    // Chi tiết lương từ dòng 7
    var dongBatDauDT = 5;       // Data nguồn từ dòng 5
    var startColTho = COL.SALARY.WORKER_START; // R = 18
    var cotBatDauLuong = COL.SALARY.START;      // N = 14
    var tongDoanhThu = 0;

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

    var response = ui.alert(
        "Xác nhận",
        "Bạn có muốn tính lương cho tháng " + sheet.getRange("A1").getValue() + "?",
        ui.ButtonSet.YES_NO
    );

    if (response != ui.Button.YES) {
        return;
    }

    // =====================================================
    // 1. CLEAR KPI CŨ I13:L
    // =====================================================
    clearThongKeNhanVienCu(sheet);

    // =====================================================
    // 2. XÁC ĐỊNH DÒNG CUỐI DATA NGUỒN THEO CỘT A
    // =====================================================
    var lastRowSheet = sheet.getLastRow();
    var dataGetLastRow = sheet.getRange("A4:A" + Math.max(lastRowSheet, 4)).getValues();
    var lastRow = dongBatDauDT - 1;

    for (var i = 0; i < dataGetLastRow.length; i++) {
        if (dataGetLastRow[i] && dataGetLastRow[i][0] !== "") {
            lastRow++;
        }
    }

    Logger.log("Dòng cuối dữ liệu nguồn: " + lastRow);

    if (lastRow < dongBatDauDT) {
        ui.alert("Không có dữ liệu bill để tính lương.");
        return;
    }

    var soDongChiTiet = lastRow - dongBatDauDT + 1;
    var dongCuoiChiTiet = dongBatDauLuong + soDongChiTiet - 1;

    // =====================================================
    // 3. CLEAR BẢNG LƯƠNG CŨ TỪ N5 TRỞ ĐI
    // =====================================================
    var lastOldOutputRow = sheet.getLastRow();

    if (lastOldOutputRow >= dongTongLuong) {
        var oldOutputRange = sheet.getRange(
            dongTongLuong,
            COL.SALARY.START,
            lastOldOutputRow - dongTongLuong + 1,
            20
        );

        breakAllMergedRanges(oldOutputRange);

        oldOutputRange
            .clearContent()
            .clearFormat()
            .setBackground("#ffffff")
            .setBorder(false, false, false, false, false, false)
            .setFontFamily("Arial")
            .setFontSize(12)
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    // =====================================================
    // 4. LẤY DỮ LIỆU NGUỒN A:G
    // =====================================================
    var dateDT = sheet.getRange(dongBatDauDT, COL.DATA.DATE, soDongChiTiet, 1).getValues();
    var customerDT = sheet.getRange(dongBatDauDT, COL.DATA.CUSTOMER, soDongChiTiet, 1).getValues();
    var billDT = sheet.getRange(dongBatDauDT, COL.DATA.BILL, soDongChiTiet, 1).getValues();
    var thoChinhDT = sheet.getRange(dongBatDauDT, COL.DATA.MAIN_STYLIST, soDongChiTiet, 1).getValues();
    var thoPhuDT = sheet.getRange(dongBatDauDT, COL.DATA.ASSISTANT, soDongChiTiet, 1).getValues();

    var columnStatusDT = sheet.getRange(
        dongBatDauDT,
        COL.DATA.SALARY_STATUS,
        soDongChiTiet,
        1
    );
    var statusDT = columnStatusDT.getValues();

    // =====================================================
    // 5. VÙNG OUTPUT CHI TIẾT N:P
    // =====================================================
    var columnDateL = sheet.getRange(dongBatDauLuong, COL.SALARY.DATE, soDongChiTiet, 1);
    var dateL = columnDateL.getValues();

    var columnCustomerL = sheet.getRange(dongBatDauLuong, COL.SALARY.CUSTOMER, soDongChiTiet, 1);
    var customerL = columnCustomerL.getValues();

    var columnBillL = sheet.getRange(dongBatDauLuong, COL.SALARY.BILL, soDongChiTiet, 1);
    columnBillL.setNumberFormat("#,##0");
    var billL = columnBillL.getValues();

    // =====================================================
    // 6. DANH SÁCH NHÂN VIÊN - CẤU HÌNH I:L
    // =====================================================
    var listThoChinh = [];
    var listThoPhu = [];

    var lastRowThoChinh = getLastNonEmptyRowInRange(
        sheet,
        COL.CONFIG.MAIN_NAME,
        COL.CONFIG.START_ROW,
        COL.CONFIG.END_ROW
    );

    var lastRowThoPhu = getLastNonEmptyRowInRange(
        sheet,
        COL.CONFIG.ASSISTANT_NAME,
        COL.CONFIG.START_ROW,
        COL.CONFIG.END_ROW
    );

    var thoChinh = [];
    var luongThoChinh = [];
    var thoPhu = [];
    var luongThoPhu = [];
    var colThoChinh = null;
    var colThoPhu = null;

    if (lastRowThoChinh >= COL.CONFIG.START_ROW) {
        var soDongThoChinh = lastRowThoChinh - COL.CONFIG.START_ROW + 1;

        colThoChinh = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.MAIN_NAME,
            soDongThoChinh,
            1
        );
        thoChinh = colThoChinh.getValues();

        luongThoChinh = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.MAIN_RATE,
            soDongThoChinh,
            1
        ).getValues();
    }

    if (lastRowThoPhu >= COL.CONFIG.START_ROW) {
        var soDongThoPhu = lastRowThoPhu - COL.CONFIG.START_ROW + 1;

        colThoPhu = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.ASSISTANT_NAME,
            soDongThoPhu,
            1
        );
        thoPhu = colThoPhu.getValues();

        luongThoPhu = sheet.getRange(
            COL.CONFIG.START_ROW,
            COL.CONFIG.ASSISTANT_RATE,
            soDongThoPhu,
            1
        ).getValues();
    }

    // Bỏ các dòng cấu hình trống, giữ rate và màu tương ứng
    var mainConfig = [];
    for (var mc = 0; mc < thoChinh.length; mc++) {
        if (thoChinh[mc][0] !== "") {
            mainConfig.push({
                name: thoChinh[mc][0],
                rate: Number(luongThoChinh[mc][0]) || 0,
                color: sheet.getRange(COL.CONFIG.START_ROW + mc, COL.CONFIG.MAIN_NAME).getBackground()
            });
        }
    }

    var assistantConfig = [];
    for (var ac = 0; ac < thoPhu.length; ac++) {
        if (thoPhu[ac][0] !== "") {
            assistantConfig.push({
                name: thoPhu[ac][0],
                rate: Number(luongThoPhu[ac][0]) || 0,
                color: sheet.getRange(COL.CONFIG.START_ROW + ac, COL.CONFIG.ASSISTANT_NAME).getBackground()
            });
        }
    }

    if (mainConfig.length === 0 && assistantConfig.length === 0) {
        ui.alert("Chưa có nhân viên trong vùng cấu hình I4:L12.");
        return;
    }

    var headerThoChinh = mainConfig.length > 0
        ? sheet.getRange(4, COL.SALARY.WORKER_START, 1, mainConfig.length).getValues()[0]
        : [];

    var headerThoPhu = assistantConfig.length > 0
        ? sheet.getRange(4, COL.SALARY.WORKER_START + mainConfig.length, 1, assistantConfig.length).getValues()[0]
        : [];

    for (var c = 0; c < mainConfig.length; c++) {
        listThoChinh.push({
            name: mainConfig[c].name,
            luong: mainConfig[c].rate,
            index: headerThoChinh.indexOf(mainConfig[c].name) + startColTho,
            color: mainConfig[c].color,
            tongLuong: 0,
            tongDoanhThuCaNhan: 0,

            // KPI tổng quát
            tongBill: 0,
            ngayLamMap: {},

            // KPI KYC
            tongDoanhThuKYC: 0,
            soBillKYC: 0,

            // KPI BSP
            tongDoanhThuBSP: 0,

            // KPI hóa chất
            soBillHoaChat: 0,
            tongDoanhThuHoaChat: 0
        });
    }

    for (var d = 0; d < assistantConfig.length; d++) {
        listThoPhu.push({
            name: assistantConfig[d].name,
            luong: assistantConfig[d].rate,
            index: headerThoPhu.indexOf(assistantConfig[d].name) + startColTho + mainConfig.length,
            color: assistantConfig[d].color,
            tongLuong: 0,
            tongDoanhThuCaNhan: 0,
            tongDoanhThuBSP: 0
        });
    }

    // =====================================================
    // 7. TÍNH TỪNG BILL
    // =====================================================
    for (var rowIndex = 0; rowIndex < soDongChiTiet; rowIndex++) {
        var ngay = dateDT[rowIndex][0];

        if (ngay === "" || ngay === null || ngay === undefined) {
            continue;
        }

        if (Number(statusDT[rowIndex][0] || 0) < 0) {
            continue;
        }

        var tenKhach = customerDT[rowIndex][0];

        if (tenKhach == "C") {
            tenKhach = "Cắt";
        } else if (tenKhach == "G") {
            tenKhach = "Gội";
        } else if (tenKhach == "CC") {
            tenKhach = "Chấm chân";
        } else if (tenKhach == "D") {
            tenKhach = "Duỗi";
        } else if (tenKhach == "U") {
            tenKhach = "Uốn";
        } else if (tenKhach == "PC") {
            tenKhach = "Phồng chân";
        } else if (tenKhach == "N") {
            tenKhach = "Nhuộm";
        }

        dateL[rowIndex][0] = ngay;
        customerL[rowIndex][0] = tenKhach;

        // Nguồn nhập đơn vị nghìn -> chuyển sang VND
        var tienBill = (Number(billDT[rowIndex][0]) || 0) * 1000;
        billL[rowIndex][0] = tienBill;

        // -------------------------
        // THỢ CHÍNH
        // -------------------------
        var currentThoChinh = listThoChinh.find(function (item) {
            return item.name == thoChinhDT[rowIndex][0];
        });

        if (currentThoChinh) {
            currentThoChinh.tongDoanhThuCaNhan += tienBill;

            // Tổng số bill của thợ chính
            currentThoChinh.tongBill += 1;

            // Ghi nhận ngày có phát sinh bill để tính số ngày làm
            var ngayLamKey = ngay instanceof Date
                ? Utilities.formatDate(
                    ngay,
                    Session.getScriptTimeZone(),
                    "dd/MM/yyyy"
                )
                : String(ngay);

            currentThoChinh.ngayLamMap[ngayLamKey] = true;

            if (isBillKYC(tenKhach)) {
                currentThoChinh.tongDoanhThuKYC += tienBill;
                currentThoChinh.soBillKYC += 1;
            }

            if (isBillBSP(tenKhach)) {
                currentThoChinh.tongDoanhThuBSP += tienBill;
            }

            if (isBillHoaChat(tenKhach)) {
                currentThoChinh.soBillHoaChat += 1;
                currentThoChinh.tongDoanhThuHoaChat += tienBill;
            }

            var luongThoChinhBill = 0;

            if (isBillKYC(tenKhach)) {
                luongThoChinhBill = tienBill / 100 * (currentThoChinh.luong + 10);
            } else if (Number(statusDT[rowIndex][0]) === 1) {
                // Giữ nguyên rule cũ: cộng % thợ phụ đầu tiên trong cấu hình
                var phanTramThoPhuMacDinh = listThoPhu.length > 0 ? listThoPhu[0].luong : 0;
                luongThoChinhBill = tienBill / 100 * (currentThoChinh.luong + phanTramThoPhuMacDinh);
            } else {
                luongThoChinhBill = tienBill / 100 * currentThoChinh.luong;
            }

            sheet.getRange(dongBatDauLuong + rowIndex, currentThoChinh.index)
                .setValue(luongThoChinhBill)
                .setFontSize(12)
                .setBackground(currentThoChinh.color)
                .setNumberFormat("#,##0");
        }

        // -------------------------
        // THỢ PHỤ
        // -------------------------
        var currentThoPhu = listThoPhu.find(function (item) {
            return item.name == thoPhuDT[rowIndex][0];
        });

        if (currentThoPhu) {
            currentThoPhu.tongDoanhThuCaNhan += tienBill;

            if (isBillBSP(tenKhach)) {
                currentThoPhu.tongDoanhThuBSP += tienBill;
            }

            var tenBillLower = String(tenKhach || "").toLowerCase();
            var luongThoPhuBill = 0;

            if (
                tenBillLower == "bsp" ||
                tenBillLower == "gội" ||
                tenBillLower == "cắt"
            ) {
                // Giữ nguyên rule cũ: dùng % của thợ chính đầu tiên trong cấu hình
                var phanTramThoChinhMacDinh = listThoChinh.length > 0 ? listThoChinh[0].luong : 0;
                luongThoPhuBill = tienBill / 100 * phanTramThoChinhMacDinh;
            } else {
                luongThoPhuBill = tienBill / 100 * currentThoPhu.luong;
            }

            sheet.getRange(dongBatDauLuong + rowIndex, currentThoPhu.index)
                .setValue(luongThoPhuBill)
                .setFontSize(12)
                .setBackground(currentThoPhu.color)
                .setNumberFormat("#,##0");
        }

        var currentDate = new Date();
        var cotNgaySua = startColTho + listThoChinh.length + listThoPhu.length;

        sheet.getRange(dongBatDauLuong + rowIndex, cotNgaySua)
            .setValue(
                Utilities.formatDate(
                    currentDate,
                    Session.getScriptTimeZone(),
                    "HH:mm dd/MM/yyyy"
                )
            )
            .setFontSize(12);
    }

    // =====================================================
    // 8. SET OUTPUT N:O:P
    // =====================================================
    columnDateL
        .setValues(dateL)
        .setFontSize(12)
        .setNumberFormat("dd/MM/yyyy");
    columnCustomerL.setValues(customerL).setFontSize(12);
    columnBillL.setValues(billL).setFontSize(12).setNumberFormat("#,##0");
    columnStatusDT.setValues(statusDT);

    // =====================================================
    // 9. TÍNH TỔNG LƯƠNG
    // =====================================================
    for (var tc = 0; tc < listThoChinh.length; tc++) {
        var listLuongChinh = sheet.getRange(
            dongBatDauLuong,
            listThoChinh[tc].index,
            soDongChiTiet,
            1
        ).getValues();

        listThoChinh[tc].tongLuong = 0;

        for (var l1 = 0; l1 < listLuongChinh.length; l1++) {
            listThoChinh[tc].tongLuong += Number(listLuongChinh[l1][0]) || 0;
        }
    }

    for (var tp = 0; tp < listThoPhu.length; tp++) {
        var listLuongPhu = sheet.getRange(
            dongBatDauLuong,
            listThoPhu[tp].index,
            soDongChiTiet,
            1
        ).getValues();

        listThoPhu[tp].tongLuong = 0;

        for (var l2 = 0; l2 < listLuongPhu.length; l2++) {
            listThoPhu[tp].tongLuong += Number(listLuongPhu[l2][0]) || 0;
        }
    }

    // =====================================================
    // 10. TỔNG DOANH THU TỪ CỘT P
    // =====================================================
    var listDoanhThu = sheet.getRange(
        dongBatDauLuong,
        COL.SALARY.BILL,
        soDongChiTiet,
        1
    ).getValues();

    for (var dt = 0; dt < listDoanhThu.length; dt++) {
        tongDoanhThu += Number(listDoanhThu[dt][0]) || 0;
    }

    // =====================================================
    // 11. HÀNG TỔNG LƯƠNG Ở DÒNG 5-6
    // N:O = Tổng | P = DT | Q = Lương | R... = Nhân viên
    // =====================================================
    sheet.getRange(dongTongLuong, COL.SALARY.DATE, 2, 2)
        .merge()
        .setValue("Tổng")
        .setFontSize(14)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    sheet.getRange(dongTongLuong, COL.SALARY.BILL, 2, 1)
        .merge()
        .setValue(tongDoanhThu)
        .setFontSize(14)
        .setFontWeight("bold")
        .setNumberFormat("#,##0")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    sheet.getRange(dongTongLuong, COL.SALARY.DAY_TOTAL, 2, 1)
        .merge()
        .setValue("Lương")
        .setFontSize(14)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    for (var sumMain = 0; sumMain < listThoChinh.length; sumMain++) {
        sheet.getRange(dongTongLuong, listThoChinh[sumMain].index, 2, 1)
            .merge()
            .setValue(listThoChinh[sumMain].tongLuong)
            .setFontSize(14)
            .setFontWeight("bold")
            .setBackground(listThoChinh[sumMain].color)
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    for (var sumAssistant = 0; sumAssistant < listThoPhu.length; sumAssistant++) {
        sheet.getRange(dongTongLuong, listThoPhu[sumAssistant].index, 2, 1)
            .merge()
            .setValue(listThoPhu[sumAssistant].tongLuong)
            .setFontSize(14)
            .setFontWeight("bold")
            .setBackground(listThoPhu[sumAssistant].color)
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    var cotNgaySuaTong = startColTho + listThoChinh.length + listThoPhu.length;

    sheet.getRange(dongTongLuong, cotNgaySuaTong, 2, 1)
        .merge()
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // =====================================================
    // 12. MERGE NGÀY + TỔNG BILL NGÀY
    // =====================================================
    var dataNgay = sheet.getRange(
        dongBatDauLuong,
        COL.SALARY.DATE,
        soDongChiTiet,
        1
    ).getValues();

    var groupStart = 0;
    var currentColor = 1;

    while (groupStart < dataNgay.length) {
        var currentDateValue = dataNgay[groupStart][0];

        if (!currentDateValue) {
            groupStart++;
            continue;
        }

        var currentDateString = Utilities.formatDate(
            currentDateValue,
            Session.getScriptTimeZone(),
            "dd/MM/yyyy"
        );

        var groupEnd = groupStart;

        while (groupEnd + 1 < dataNgay.length && dataNgay[groupEnd + 1][0]) {
            var nextDateString = Utilities.formatDate(
                dataNgay[groupEnd + 1][0],
                Session.getScriptTimeZone(),
                "dd/MM/yyyy"
            );

            if (nextDateString !== currentDateString) {
                break;
            }

            groupEnd++;
        }

        var groupLength = groupEnd - groupStart + 1;
        var sheetStartRow = dongBatDauLuong + groupStart;

        var dateRange = sheet.getRange(
            sheetStartRow,
            COL.SALARY.DATE,
            groupLength,
            1
        );

        if (groupLength > 1) {
            var dateValue = dateRange.getCell(1, 1).getValue();
            dateRange.clearContent().merge().setValue(dateValue);
        }

        dateRange
            .setFontWeight("bold")
            .setFontSize(12)
            .setNumberFormat("dd/MM/yyyy")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");

        var billsInDay = sheet.getRange(
            sheetStartRow,
            COL.SALARY.BILL,
            groupLength,
            1
        ).getValues();

        var totalDay = 0;
        for (var b = 0; b < billsInDay.length; b++) {
            totalDay += Number(billsInDay[b][0]) || 0;
        }

        var totalDayRange = sheet.getRange(
            sheetStartRow,
            COL.SALARY.DAY_TOTAL,
            groupLength,
            1
        );

        if (groupLength > 1) {
            totalDayRange.clearContent().merge().setValue(totalDay);
        } else {
            totalDayRange.setValue(totalDay);
        }

        totalDayRange
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");

        var hangColor;
        if (currentColor === 1) {
            hangColor = "#d9d2e9";
            currentColor = 2;
        } else {
            hangColor = "#ffffff";
            currentColor = 1;
        }

        sheet.getRange(
            sheetStartRow,
            COL.SALARY.DATE,
            groupLength,
            4
        ).setBackground(hangColor);

        groupStart = groupEnd + 1;
    }

    // =====================================================
    // HIGHLIGHT KYC / BSP
    // PHẢI ĐỂ SAU TÔ MÀU NGÀY
    // =====================================================

    highlightSpecialBills(
        sheet,
        dongBatDauLuong,
        customerL
    );

    // =====================================================
    // 13. BORDER TOÀN BỘ BẢNG LƯƠNG
    // =====================================================
    var soCotBangLuong = 5 + listThoChinh.length + listThoPhu.length;
    var soDongBangLuong = dongCuoiChiTiet - dongTongLuong + 1;

    sheet.getRange(
        dongTongLuong,
        cotBatDauLuong,
        soDongBangLuong,
        soCotBangLuong
    )
        .setBorder(true, true, true, true, true, true)
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // =====================================================
    // 14. KPI CỐ ĐỊNH I13:L + CẢNH BÁO QUẢN TRỊ
    // =====================================================
    xuatThongKeNhanVien(
        sheet,
        COL.KPI.START_ROW,
        listThoChinh,
        listThoPhu
    );

    // =====================================================
    // 15. BẢNG TỔNG QUÁT THỢ CHÍNH
    // Bắt đầu từ cột N, dưới bảng lương 1 hàng trống; có thêm cột Trạng thái
    // =====================================================
    var dongBatDauBangTongQuat = dongCuoiChiTiet + 2;

    xuatBangTongQuatThoChinh(
        sheet,
        dongBatDauBangTongQuat,
        listThoChinh,
        tongDoanhThu
    );
}

function filterMultipleColumns() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

    if (sheet.getFilter()) {
        sheet.getFilter().remove();
    }

    // Sau khi bỏ cột D, vùng cũ S:U dịch sang R:T
    var range = sheet.getRange("R4:T");
    var filter = range.createFilter();

    filter.setColumnFilterCriteria(
        20, // cột T
        SpreadsheetApp.newFilterCriteria()
            .whenNumberBetween(0, 500000000000000000)
            .build()
    );
}

function sheetToPDF(sheet, range) {
    var spreadsheet = sheet.getParent();
    var sheetId = sheet.getSheetId();
    var rangeA1 = range.getA1Notation();

    var url = `https://docs.google.com/spreadsheets/d/${spreadsheet.getId()}/export?`;
    url += `format=pdf&gid=${sheetId}&range=${rangeA1}`;
    url += "&portrait=true&size=A4";

    var params = {
        muteHttpExceptions: true,
        headers: {
            "Authorization": "Bearer " + ScriptApp.getOAuthToken()
        }
    };

    var response = UrlFetchApp.fetch(url, params);

    return response.getBlob().setName("Exported_PDF.pdf");
}

function getRandomPastelColorForTemplate(usedColors) {

    var colors = [
        "#FFF2CC", // vàng kem
        "#D9EAD3", // xanh lá nhạt
        "#CFE2F3", // xanh dương nhạt
        "#D9D2E9", // tím nhạt
        "#EAD1DC", // hồng tím nhạt
        "#FCE5CD", // cam kem
        "#F4CCCC", // đỏ hồng nhạt
        "#D0E0E3", // xanh xám nhạt
        "#E2F0D9",
        "#DDEBF7",
        "#E4DFEC",
        "#FCE4D6",
        "#FFF4CC",
        "#E8DAEF",
        "#D6EAF8",
        "#D5F5E3",
        "#FCF3CF"
    ];

    var availableColors = colors.filter(function (color) {
        return !usedColors.includes(color);
    });

    if (availableColors.length === 0) {
        availableColors = colors;
    }

    var color = availableColors[
        Math.floor(Math.random() * availableColors.length)
    ];

    usedColors.push(color);

    return color;
}


function getUniqueWorkerColors() {

    // 12 màu khác nhau, nổi bật nhưng vẫn đủ sáng cho chữ đen
    var colors = [
        "#FFD966", // vàng
        "#93C47D", // xanh lá
        "#76A5AF", // xanh ngọc
        "#6FA8DC", // xanh dương
        "#8E7CC3", // tím
        "#C27BA0", // hồng tím

        "#F6B26B", // cam
        "#E06666", // đỏ san hô
        "#A2C4C9", // xanh xám
        "#B6D7A8", // xanh lá sáng
        "#A4C2F4", // xanh lam tím
        "#D5A6BD"  // hồng
    ];

    // Shuffle để mỗi lần tạo sheet thứ tự màu khác nhau
    for (var i = colors.length - 1; i > 0; i--) {

        var j = Math.floor(
            Math.random() * (i + 1)
        );

        var temp = colors[i];
        colors[i] = colors[j];
        colors[j] = temp;
    }

    return colors;
}


function createTables() {

    var ui = SpreadsheetApp.getUi();

    var response = ui.alert(
        "Xác nhận",
        "Có muốn tạo bảng mẫu? Lưu ý nên tạo sheet mới trước!",

        "Thợ phụ",
        "Ghi chú",
        "Tính lương"

    for (
        var col = 1;
        col <= 7;
        col++

            .setBorder(

                true,
                true,
                true
            );
    }

    sheet.getRange("I1:L1")
        .merge()
        .setValue("Nhân viên")
        .setFontWeight("bold")
        .setFontSize(13)
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle");

    // 5. HEADER CONFIG NHÂN VIÊN
    // =====================================================

    sheet.getRange("I3:J3")

            "Thợ chính",
            "Lương (%)"
        ]])

    // 6 THỢ CHÍNH + 6 THỢ PHỤ

    var startWorkerRow = 4;
    var workerRows = 6;

    var mainWorkerRange =
        sheet.getRange(
            startWorkerRow,
            9,
            workerRows,
            1
        );

        sheet.getRange(
            startWorkerRow,
            10,
            workerRows,

        );

        sheet.getRange(
            startWorkerRow,
            11,
            workerRows,

        sheet.getRange(

        getUniqueWorkerColors();

    // 6 THỢ CHÍNH
    // -----------------------------

    for (
        var i = 0;
        i < workerRows;
        i++
    ) {

        mainWorkerRange
            .getCell(
                i + 1,
                1
            )
            .setBackground(
                workerColors[i]
            )
            .setFontColor("#000000")
            .setFontWeight("bold")
            .setFontSize(12)
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }


    // -----------------------------
    // 6 THỢ PHỤ
    // -----------------------------

    for (
        var i = 0;
        i < workerRows;
        i++
    ) {

        assistantWorkerRange
            .getCell(
                i + 1,

                workerColors[i + 6]

    mainRateRange
        .setNumberFormat("0.##")
        .setFontSize(12)
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");


    assistantRateRange
        .setNumberFormat("0.##")
        .setFontSize(12)
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");


    // =====================================================
    // 9. BORDER BẢNG NHÂN VIÊN
    // I3:L9
    // =====================================================

    sheet.getRange(
        3,
        9,
        workerRows + 1,
        4
    )
        .setBorder(
            true,
            true,
            true,
            true,
            true,
            true
        )
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");


    // =====================================================
    // 10. VALIDATION THỢ CHÍNH
    // =====================================================

    var validationThoChinh =
        SpreadsheetApp
            .newDataValidation()
            .requireValueInRange(
                mainWorkerRange,
                true
            )
            .setAllowInvalid(false)
            .build();


    // =====================================================
    // 11. VALIDATION THỢ PHỤ
    // =====================================================

    var validationThoPhu =
        SpreadsheetApp
            .newDataValidation()
            .requireValueInRange(
                assistantWorkerRange,
                true
            )
            .setAllowInvalid(false)
            .build();


    // =====================================================
    // 12. SET DROPDOWN CHO BẢNG NHẬP
    // =====================================================

    // D = thợ chính
    sheet.getRange("D5:D500")
        .setDataValidation(
            validationThoChinh
        );


    // E = thợ phụ
    sheet.getRange("E5:E500")
        .setDataValidation(
            validationThoPhu
        );


    // =====================================================
    // 13. FORMAT BẢNG NHẬP
    // =====================================================

    sheet.getRange("A5:G500")
        .setFontFamily("Arial")
        .setFontSize(12)
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");


    // Border
    sheet.getRange("A4:G500")
        .setBorder(
            true,
            true,
            true,
            true,
            true,
            true
        );


    // =====================================================
    // 14. FORMAT CỘT
    // =====================================================

    // A - Ngày
    sheet.setColumnWidth(
        1,
        110
    );

    // B - Tên khách
    sheet.setColumnWidth(
        2,
        140
    );

    // C - Tiền bill
    sheet.setColumnWidth(
        3,
        100
    );

    // D - Thợ chính
    sheet.setColumnWidth(
        4,
        110
    );

    // E - Thợ phụ
    sheet.setColumnWidth(
        5,
        110
    );

    // F - Ghi chú
    sheet.setColumnWidth(
        6,
        140
    );

    // G - Tính lương
    sheet.setColumnWidth(
        7,
        100
    );


    // =====================================================
    // 15. WIDTH CONFIG NHÂN VIÊN
    // =====================================================

    sheet.setColumnWidth(
        9,
        115
    ); // I

    sheet.setColumnWidth(
        10,
        100
    ); // J

    sheet.setColumnWidth(
        11,
        115
    ); // K

    sheet.setColumnWidth(
        12,
        100
    ); // L


    // =====================================================
    // 16. FORMAT TIỀN BILL
    // =====================================================

    sheet.getRange("C5:C500")
        .setNumberFormat("#,##0");


    // =====================================================
    // 17. FORMAT NGÀY
    // =====================================================

    sheet.getRange("A5:A500")
        .setNumberFormat(
            "dd/MM/yyyy"
        );


    // =====================================================
    // 18. FREEZE HEADER
    // =====================================================

    sheet.setFrozenRows(4);


    // =====================================================
    // 19. HOÀN TẤT
    // =====================================================

    ui.alert(
        "Tạo sheet mẫu thành công."
    );
}

function runWithProgress() {
    var html = HtmlService.createHtmlOutputFromFile("progress")
        .setWidth(400)
        .setHeight(150);

    SpreadsheetApp.getUi().showModalDialog(
        html,
        "Đang xử lý..."
    );

    for (var i = 0; i <= 100; i += 10) {
        updateProgressInHtml(i);
        Utilities.sleep(300);
    }

    setThangDefault();
    updateProgressInHtml(100);
}

function updateProgressInHtml(value) {
    var template = HtmlService.createHtmlOutput(
        "<script>window.parent.updateProgress(" +
        value +
        ");</script>"
    );

    SpreadsheetApp.getUi().showModalDialog(
        template,
        "Cập nhật tiến trình..."
    );
}

function exportRangeToPDF() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var spreadsheetId = SpreadsheetApp.getActiveSpreadsheet().getId();
    var sheetId = sheet.getSheetId();

    var startRow = 3;
    var startCol = COL.SALARY.START; // N = 14

    var lastRow = sheet.getLastRow() + 1;
    var lastCol = sheet.getLastColumn();

    var numRows = lastRow - startRow + 1;
    var numCols = lastCol - startCol + 1;

    var rangeA1 = sheet.getRange(startRow, startCol, numRows, numCols).getA1Notation();

    var url = "https://docs.google.com/spreadsheets/d/" + spreadsheetId + "/export?format=pdf" +
        "&portrait=true" +
        "&size=A4" +
        "&gid=" + sheetId +
        "&range=" + rangeA1 +
        "&gridlines=false" +
        "&fitw=true";

    var token = ScriptApp.getOAuthToken();
    var response = UrlFetchApp.fetch(url, {
        headers: {
            "Authorization": "Bearer " + token
        }
    });

    var pdfBlob = response.getBlob().setName("Bảng lương Thái Bình.pdf");
    var file = DriveApp.createFile(pdfBlob);
    var downloadUrl = file.getDownloadUrl();

    var html = '<script>window.open("' + downloadUrl + '", "_blank"); google.script.host.close();</script>';
    var ui = HtmlService.createHtmlOutput(html);

    SpreadsheetApp.getUi().showModalDialog(ui, "Đang tải xuống...");
}
