function normalizeTenBill(value) {
    return String(value || "").trim().toUpperCase();
}

function isBillKYC(value) {
    return normalizeTenBill(value).includes("KYC");
}

function isBillBSP(value) {
    return normalizeTenBill(value) === "BSP";
}

function isBillHoaChat(value) {
    var tenBill = normalizeTenBill(value);

    if (!tenBill) {
        return false;
    }

    // Không tính các bill dịch vụ này là bill hóa chất
    var exclude = [
        "C",
        "CẮT",
        "CAT",
        "G",
        "GỘI",
        "GOI",
        "DVL",
        "BSP"
    ];

    return !exclude.includes(tenBill);
}

function breakAllMergedRanges(range) {
    var mergedRanges = range.getMergedRanges();

    for (var i = 0; i < mergedRanges.length; i++) {
        mergedRanges[i].breakApart();
    }
}

function clearThongKeNhanVienCu(sheet) {

    // Bảng thống kê mới cố định trong J13:M
    var startRow = 13;
    var startCol = 10; // J
    var numCols = 4;

    // Clear khoảng đủ lớn để chứa thống kê
    // Ví dụ J13:M60
    var numRows = 48;

    var range = sheet.getRange(
        startRow,
        startCol,
        numRows,
        numCols
    );

    // Phải break merge trước
    range.breakApart();

    range
        .clearContent()
        .clearFormat()
        .setBackground("#ffffff")
        .setBorder(
            false,
            false,
            false,
            false,
            false,
            false
        )
        .setFontFamily("Arial")
        .setFontSize(13);
}

                tho.tongLuong /

            rowTongQuanData,
            startCol,
            tongQuanRows.length,
            totalCols

            .setValues(tongQuanRows)
            .setFontSize(fontSize)

            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);

            rowTongQuanData,
            startCol + 1,
            tongQuanRows.length,
            2
        ).setNumberFormat("#,##0");

        sheet.getRange(
            rowTongQuanData,
            startCol + 3,
            tongQuanRows.length,

    // TÔ MÀU + IN ĐẬM TÊN NHÂN VIÊN Ở TỔNG QUAN
    // =====================================================

        sheet.getRange(
            rowTongQuanData + i,
            startCol
        )
            .setBackground(listThoChinh[i].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }

    for (var i = 0; i < listThoPhu.length; i++) {

        sheet.getRange(
            rowTongQuanData + listThoChinh.length + i,
            startCol

            .setBackground(listThoPhu[i].color)
            .setFontWeight("bold")

            .setHorizontalAlignment("left");
    }

    // =====================================================

    var rowDoanhThuTitle =
        rowTongQuanData +

        1;

    var rowDoanhThuHeader =
        rowDoanhThuTitle + 1;

        rowDoanhThuHeader + 1;

        rowDoanhThuTitle,
        startCol,
        1,
        totalCols
    )

        .setValue("DOANH THU ĐẶC BIỆT - THỢ CHÍNH")
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")
        .setBackground("#eeeeee")
        .setBorder(true, true, true, true, true, true);

    var headerDoanhThu = [[
        "Nhân viên",
        "DT KYC",
        "DT BSP",
        "Lương / DT"
    ]];

    sheet.getRange(
        rowDoanhThuHeader,
        startCol,
        1,
        totalCols
    )
        .setValues(headerDoanhThu)
        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle")
        .setBackground("#d3d3d3")
        .setBorder(true, true, true, true, true, true);

    var doanhThuRows = [];

    listThoChinh.forEach(function (tho) {

        var tyLeLuong = 0;

        if (tho.tongDoanhThuCaNhan > 0) {
            tyLeLuong =
                tho.tongLuong /
                tho.tongDoanhThuCaNhan;

        doanhThuRows.push([
            tho.name,
            tho.tongDoanhThuKYC,
            tho.tongDoanhThuBSP,

    if (doanhThuRows.length > 0) {

        sheet.getRange(
            rowDoanhThuData,
            startCol,
            doanhThuRows.length,
            totalCols
        )
            .setValues(doanhThuRows)
            .setFontSize(fontSize)
            .setFontWeight("normal")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")
            .setBorder(true, true, true, true, true, true);

        // KYC + BSP
        sheet.getRange(
            rowDoanhThuData,
            startCol + 1,
            doanhThuRows.length,
            2
        ).setNumberFormat("#,##0");

        // %
        sheet.getRange(
            rowDoanhThuData,
            startCol + 3,
            doanhThuRows.length,
            1
        ).setNumberFormat("0.00%");
    }

    // Màu tên thợ chính
    for (var i = 0; i < listThoChinh.length; i++) {

        sheet.getRange(
            rowDoanhThuData + i,
            startCol
        )
            .setBackground(listThoChinh[i].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");

        startCol,

        .setFontSize(fontSize)
        .setFontWeight("bold")
        .setHorizontalAlignment("left")
        .setVerticalAlignment("middle")
        .setBackground("#eeeeee")

        "SL bill HC",
        "DT hóa chất",
        "Bill TB"

    sheet.getRange(
        rowHoaChatHeader,
        startCol,
        1,
        totalCols

        if (tho.soBillHoaChat > 0) {

                tho.tongDoanhThuHoaChat /

            tho.soBillHoaChat,

            rowHoaChatData,

            hoaChatRows.length,
            totalCols
        )

            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle")

            rowHoaChatData,
            startCol + 1,

        sheet.getRange(
            rowHoaChatData,
            startCol + 2,
            hoaChatRows.length,

    for (var i = 0; i < listThoChinh.length; i++) {

        sheet.getRange(
            rowHoaChatData + i,
            startCol
        )
            .setBackground(listThoChinh[i].color)
            .setFontWeight("bold")
            .setFontSize(fontSize)
            .setHorizontalAlignment("left");
    }
}

        if (lastColumn > 17) {

            .setColumnWidth(21, 150)
            .setColumnWidth(22, 150)

            .setColumnWidth(24, 150)
            .setColumnWidth(25, 150)
            .setColumnWidth(26, 150);

        const cellTitle = sheet.getRange("O1");

        cellTitle.setFontSize(17).setHorizontalAlignment("left");

            sheet.getRange(3, col).setValue(values[0][col - 15]);
            sheet.getRange(3, col).setVerticalAlignment("middle").setHorizontalAlignment("center");

        headerCommons.setFontWeight("bold");
        headerCommons.setFontSize(12);
        headerCommons.setBackground("#d3d3d3");
        headerCommons.setBorder(true, true, true, true, true, true);

        let lastRowThoChinh = sheet.getRange(sheet.getMaxRows(), 10).getNextDataCell(SpreadsheetApp.Direction.UP).getRow();
        let lastRowThoPhu = sheet.getRange(sheet.getMaxRows(), 12).getNextDataCell(SpreadsheetApp.Direction.UP).getRow();

        let colThoChinh = sheet.getRange("J4:J" + lastRowThoChinh);
        let thoChinh = colThoChinh.getValues();

        let colThoPhu = sheet.getRange("L4:L" + lastRowThoPhu);
        let thoPhu = colThoPhu.getValues();

        var startCellTho = sheet.getRange("S4");
        let coutTho = 0;

        for (let i = 0; i < thoChinh.length; i++) {
            startCellTho.offset(0, coutTho).setValue(thoChinh[i])
                .setFontWeight("bold")
                .setFontSize(12)
                .setBackground(colThoChinh.getCell(i + 1, 1).getBackground())
                .setBorder(true, true, true, true, true, true)
                .setHorizontalAlignment("center");
            coutTho += 1;
        }

        sheet.getRange(3, 19, 1, thoChinh.length).merge();
        sheet.getRange(3, 19).setValue("Thợ chính")
            .setFontWeight("bold")
            .setFontSize(12)
            .setBackground("#d3d3d3")
            .setBorder(true, true, true, true, true, true)
            .setHorizontalAlignment("center");

        for (let i = 0; i < thoPhu.length; i++) {
            startCellTho.offset(0, coutTho).setValue(thoPhu[i])
                .setFontWeight("bold")
                .setFontSize(12)
                .setBackground(colThoPhu.getCell(i + 1, 1).getBackground())
                .setBorder(true, true, true, true, true, true)
                .setHorizontalAlignment("center");
            coutTho += 1;

        sheet.getRange(3, 19 + thoChinh.length, 1, thoPhu.length).merge();

            .setFontWeight("bold")

            .setBorder(true, true, true, true, true, true)

        sheet.getRange(3, 19 + thoChinh.length + thoPhu.length, 2, 1).merge()

function tinhLuong() {

    var ui = SpreadsheetApp.getUi();

    // =====================================================
    // CẤU HÌNH VỊ TRÍ
    // =====================================================

    // Tổng lương nằm tại dòng 5-6
    let dongTongLuong = 5;

    // Danh sách bill bắt đầu từ dòng 7
    let dongBatDauLuong = 7;

    // Dữ liệu nguồn vẫn bắt đầu từ dòng 5
    let dongBatDauDT = 5;

    // Cột thợ bắt đầu từ S = 19
    let startColTho = 19;

    // Bảng lương bắt đầu từ O = 15
    let cotBatDauLuong = 15;

    let tongDoanhThu = 0;

    var sheet = SpreadsheetApp
        .getActiveSpreadsheet()
        .getActiveSheet();

    // =====================================================
    // HÀM NỘI BỘ NHẬN DIỆN BILL
    // =====================================================

    function normalizeBill(value) {
        return String(value || "")
            .trim()
            .toUpperCase();
    }

    function billKYC(value) {
        return normalizeBill(value).includes("KYC");
    }

    function billBSP(value) {
        return normalizeBill(value) === "BSP";
    }

    function billHoaChat(value) {

        var tenBill = normalizeBill(value);

        if (!tenBill) {
            return false;
        }

        // Không tính các bill này là hóa chất
        var exclude = [
            "C",
            "CẮT",
            "CAT",

            "G",
            "GỘI",
            "GOI",

            "DVL",
            "BSP"
        ];

        return !exclude.includes(tenBill);
    }

    // =====================================================
    // CONFIRM
    // =====================================================

    var response = ui.alert(
        "Xác nhận",
        "Bạn có muốn tính lương cho tháng " +
            sheet.getRange("A1").getValue() +
            "?",
        ui.ButtonSet.YES_NO
    );

    if (response != ui.Button.YES) {
        return;
    }

    // =====================================================
    // 1. CLEAR BẢNG THỐNG KÊ KPI CŨ J13:M
    // =====================================================

    var rangeThongKeCu = sheet.getRange(
        13,     // row
        10,     // J
        50,     // đủ vùng thống kê
        4       // J:M
    );

    // Unmerge toàn bộ merge trong vùng KPI
    var mergedThongKe =
        rangeThongKeCu.getMergedRanges();

    for (
        var mt = 0;
        mt < mergedThongKe.length;
        mt++
    ) {
        mergedThongKe[mt].breakApart();
    }

    rangeThongKeCu
        .clearContent()
        .clearFormat()
        .setBackground("#ffffff")
        .setBorder(
            false,
            false,
            false,
            false,
            false,
            false
        )
        .setFontFamily("Arial")
        .setFontSize(13)
        .setFontWeight("normal");

    // =====================================================
    // 2. XÁC ĐỊNH DÒNG CUỐI DỮ LIỆU NGUỒN
    // Dựa theo cột A
    // =====================================================

    var lastRowSheet = sheet.getLastRow();

    var dataGetLastRow = sheet.getRange(
        "A4:A" + lastRowSheet
    ).getValues();

    var lastRow = dongBatDauDT - 1;

    for (
        var i = 0;
        i < dataGetLastRow.length;
        i++
    ) {

        if (
            dataGetLastRow[i] &&
            dataGetLastRow[i][0] !== ""
        ) {
            lastRow++;
        }
    }

    Logger.log(
        "Dòng cuối dữ liệu nguồn: " +
        lastRow
    );

    if (lastRow < dongBatDauDT) {

        ui.alert(
            "Không có dữ liệu bill để tính lương."
        );

        return;
    }

    // Tổng số bill / dòng dữ liệu
    var soDongChiTiet =
        lastRow - dongBatDauDT + 1;

    // Vì output bắt đầu từ dòng 7
    var dongCuoiChiTiet =
        dongBatDauLuong +
        soDongChiTiet -
        1;

    // =====================================================
    // 3. CLEAR BẢNG LƯƠNG CŨ
    // =====================================================

    /*
       Clear từ O5 trở xuống.

       Quan trọng:
       phải getMergedRanges()
       rồi breakApart từng vùng merge đầy đủ.

       Không break riêng cột O hoặc R nữa.
    */

    var lastOldOutputRow =
        sheet.getLastRow();

    if (lastOldOutputRow >= dongTongLuong) {

        var oldOutputRange = sheet.getRange(
            dongTongLuong,
            15, // O
            lastOldOutputRow -
                dongTongLuong +
                1,
            20 // O -> AH
        );

        var mergedRanges =
            oldOutputRange.getMergedRanges();

        for (
            var m = 0;
            m < mergedRanges.length;
            m++
        ) {
            mergedRanges[m].breakApart();
        }

        oldOutputRange
            .clearContent()
            .clearFormat()
            .setBackground("#ffffff")
            .setBorder(
                false,
                false,
                false,
                false,
                false,
                false
            )
            .setFontFamily("Arial")
            .setFontSize(12)
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    // =====================================================
    // 4. LẤY DỮ LIỆU DOANH THU A:H
    // =====================================================

    // Ngày
    const columnDateDT = sheet.getRange(
        dongBatDauDT,
        1,
        soDongChiTiet,
        1
    );

    const dateDT =
        columnDateDT.getValues();

    // Tên khách / loại bill
    const columnCustomerDT = sheet.getRange(
        dongBatDauDT,
        2,
        soDongChiTiet,
        1
    );

    const customerDT =
        columnCustomerDT.getValues();

    // Tiền bill
    const columnBillDT = sheet.getRange(
        dongBatDauDT,
        3,
        soDongChiTiet,
        1
    );

    const billDT =
        columnBillDT.getValues();

    // Thợ chính - E
    const columnThoChinhDT = sheet.getRange(
        dongBatDauDT,
        5,
        soDongChiTiet,
        1
    );

    const thoChinhDT =
        columnThoChinhDT.getValues();

    // Thợ phụ - F
    const columnThoPhuDT = sheet.getRange(
        dongBatDauDT,
        6,
        soDongChiTiet,
        1
    );

    const thoPhuDT =
        columnThoPhuDT.getValues();

    // Tính lương - H
    const columnStatusDT = sheet.getRange(
        dongBatDauDT,
        8,
        soDongChiTiet,
        1
    );

    const statusDT =
        columnStatusDT.getValues();

    // =====================================================
    // 5. VÙNG OUTPUT CHI TIẾT
    // BẮT ĐẦU TỪ DÒNG 7
    // =====================================================

    // Ngày - O
    const columnDateL = sheet.getRange(
        dongBatDauLuong,
        15,
        soDongChiTiet,
        1
    );

    const dateL =
        columnDateL.getValues();

    // Tên khách - P
    const columnCustomerL = sheet.getRange(
        dongBatDauLuong,
        16,
        soDongChiTiet,
        1
    );

    const customerL =
        columnCustomerL.getValues();

    // Tiền bill - Q
    const columnBillL = sheet.getRange(
        dongBatDauLuong,
        17,
        soDongChiTiet,
        1
    );

    columnBillL.setNumberFormat("#,##0");

    const billL =
        columnBillL.getValues();

    // =====================================================
    // 6. DANH SÁCH NHÂN VIÊN
    // =====================================================

    let listThoChinh = [];
    let listThoPhu = [];

    // -------------------------
    // THỢ CHÍNH
    // -------------------------

    let lastRowThoChinh =
        sheet.getRange(
            sheet.getMaxRows(),
            10
        )
        .getNextDataCell(
            SpreadsheetApp.Direction.UP
        )
        .getRow();

    let colThoChinh = sheet.getRange(
        "J4:J" + lastRowThoChinh
    );

    let thoChinh =
        colThoChinh.getValues();

    let colLuongThoChinh =
        sheet.getRange(
            "K4:K" + lastRowThoChinh
        );

    let luongThoChinh =
        colLuongThoChinh.getValues();

    // -------------------------
    // THỢ PHỤ
    // -------------------------

    let lastRowThoPhu =
        sheet.getRange(
            sheet.getMaxRows(),
            12
        )
        .getNextDataCell(
            SpreadsheetApp.Direction.UP
        )
        .getRow();

    let colThoPhu = sheet.getRange(
        "L4:L" + lastRowThoPhu
    );

    let thoPhu =
        colThoPhu.getValues();

    let colLuongThoPhu =
        sheet.getRange(
            "M4:M" + lastRowThoPhu
        );

    let luongThoPhu =
        colLuongThoPhu.getValues();

    // Header bảng lương
    var headerThoChinh =
        sheet.getRange(
            4,
            19,
            1,
            thoChinh.length
        )
        .getValues()[0];

    var headerThoPhu =
        sheet.getRange(
            4,
            19 + thoChinh.length,
            1,
            thoPhu.length
        )
        .getValues()[0];

    // =====================================================
    // 7. KHỞI TẠO THỢ CHÍNH + KPI
    // =====================================================

    for (
        let i = 0;
        i < thoChinh.length;
        i++
    ) {

        listThoChinh.push({

            name:
                thoChinh[i][0],

            luong:
                Number(
                    luongThoChinh[i][0]
                ) || 0,

            index:
                headerThoChinh.indexOf(
                    thoChinh[i][0]
                ) +
                startColTho,

            color:
                colThoChinh
                    .getCell(
                        i + 1,
                        1
                    )
                    .getBackground(),

            // Lương
            tongLuong: 0,

            // KPI
            tongDoanhThuCaNhan: 0,

            tongDoanhThuKYC: 0,

            tongDoanhThuBSP: 0,

            soBillHoaChat: 0,

            tongDoanhThuHoaChat: 0
        });
    }

    // =====================================================
    // 8. KHỞI TẠO THỢ PHỤ + KPI
    // =====================================================

    for (
        let i = 0;
        i < thoPhu.length;
        i++
    ) {

        listThoPhu.push({

            name:
                thoPhu[i][0],

            luong:
                Number(
                    luongThoPhu[i][0]
                ) || 0,

            index:
                headerThoPhu.indexOf(
                    thoPhu[i][0]
                ) +
                startColTho +
                thoChinh.length,

            color:
                colThoPhu
                    .getCell(
                        i + 1,
                        1
                    )
                    .getBackground(),

            tongLuong: 0,

            tongDoanhThuCaNhan: 0,

            tongDoanhThuBSP: 0
        });
    }

    // =====================================================
    // 9. TÍNH TỪNG BILL
    // =====================================================

    for (
        let i = 0;
        i < soDongChiTiet;
        i++
    ) {

        var ngay =
            dateDT[i][0];

        if (
            ngay === "" ||
            ngay === null ||
            ngay === undefined
        ) {
            continue;
        }

        /*
         * Giữ đúng logic cũ:
         * blank H cũng được hiểu tương đương 0.
         */
        if (
            Number(
                statusDT[i][0] || 0
            ) < 0
        ) {
            continue;
        }

        // =================================================
        // CHUẨN HÓA TÊN BILL HIỂN THỊ
        // =================================================

        var tenKhach =
            customerDT[i][0];

        if (tenKhach == "C") {

            tenKhach = "Cắt";

        } else if (tenKhach == "G") {

            tenKhach = "Gội";

        } else if (tenKhach == "CC") {

            tenKhach = "Chấm chân";

        } else if (tenKhach == "D") {

            tenKhach = "Duỗi";

        } else if (tenKhach == "U") {

            tenKhach = "Uốn";

        } else if (tenKhach == "PC") {

            tenKhach = "Phồng chân";

        } else if (tenKhach == "N") {

            tenKhach = "Nhuộm";
        }

        // =================================================
        // OUTPUT THÔNG TIN CƠ BẢN
        // =================================================

        dateL[i][0] = ngay;

        customerL[i][0] =
            tenKhach;

        // Tiền bill nguồn đang nhập đơn vị nghìn
        var tienBill =
            (
                Number(
                    billDT[i][0]
                ) || 0
            ) *
            1000;

        billL[i][0] =
            tienBill;

        // =================================================
        // THỢ CHÍNH
        // =================================================

        const curentThoChinh =
            listThoChinh.find(
                item =>
                    item.name ==
                    thoChinhDT[i][0]
            );

        if (curentThoChinh) {

            // -------------------------
            // KPI doanh thu riêng
            // -------------------------

            curentThoChinh
                .tongDoanhThuCaNhan +=
                tienBill;

            // KYC
            if (billKYC(tenKhach)) {

                curentThoChinh
                    .tongDoanhThuKYC +=
                    tienBill;
            }

            // BSP
            if (billBSP(tenKhach)) {

                curentThoChinh
                    .tongDoanhThuBSP +=
                    tienBill;
            }

            // Hóa chất
            if (billHoaChat(tenKhach)) {

                curentThoChinh
                    .soBillHoaChat += 1;

                curentThoChinh
                    .tongDoanhThuHoaChat +=
                    tienBill;
            }

            // =================================================
            // TÍNH LƯƠNG THỢ CHÍNH
            // =================================================

            var luongThoChinhBill = 0;

            // KYC
            if (billKYC(tenKhach)) {

                luongThoChinhBill =
                    tienBill /
                    100 *
                    (
                        curentThoChinh.luong +
                        10
                    );

            // H = 1
            } else if (
                Number(
                    statusDT[i][0]
                ) === 1
            ) {

                /*
                 * GIỮ NGUYÊN RULE CŨ:
                 *
                 * % thợ chính
                 * +
                 * % thợ phụ đầu tiên
                 */
                var phanTramThoPhuMacDinh =
                    listThoPhu.length > 0
                        ? listThoPhu[0].luong
                        : 0;

                luongThoChinhBill =
                    tienBill /
                    100 *
                    (
                        curentThoChinh.luong +
                        phanTramThoPhuMacDinh
                    );

            } else {

                luongThoChinhBill =
                    tienBill /
                    100 *
                    curentThoChinh.luong;
            }

            sheet.getRange(
                dongBatDauLuong + i,
                curentThoChinh.index
            )
                .setValue(
                    luongThoChinhBill
                )
                .setFontSize(12)
                .setBackground(
                    curentThoChinh.color
                )
                .setNumberFormat(
                    "#,##0"
                );
        }

        // =================================================
        // THỢ PHỤ
        // =================================================

        const curentThoPhu =
            listThoPhu.find(
                item =>
                    item.name ==
                    thoPhuDT[i][0]
            );

        if (curentThoPhu) {

            // Doanh thu riêng
            curentThoPhu
                .tongDoanhThuCaNhan +=
                tienBill;

            // BSP
            if (billBSP(tenKhach)) {

                curentThoPhu
                    .tongDoanhThuBSP +=
                    tienBill;
            }

            // =================================================
            // TÍNH LƯƠNG THỢ PHỤ
            // =================================================

            var tenBillLower =
                String(
                    tenKhach || ""
                )
                .toLowerCase();

            var luongThoPhuBill = 0;

            if (
                tenBillLower == "bsp" ||
                tenBillLower == "gội" ||
                tenBillLower == "cắt"
            ) {

                /*
                 * GIỮ NGUYÊN RULE CŨ:
                 *
                 * BSP/Gội/Cắt thợ phụ
                 * lấy % thợ chính đầu tiên.
                 */
                var phanTramThoChinhMacDinh =
                    listThoChinh.length > 0
                        ? listThoChinh[0].luong
                        : 0;

                luongThoPhuBill =
                    tienBill /
                    100 *
                    phanTramThoChinhMacDinh;

            } else {

                luongThoPhuBill =
                    tienBill /
                    100 *
                    curentThoPhu.luong;
            }

            sheet.getRange(
                dongBatDauLuong + i,
                curentThoPhu.index
            )
                .setValue(
                    luongThoPhuBill
                )
                .setFontSize(12)
                .setBackground(
                    curentThoPhu.color
                )
                .setNumberFormat(
                    "#,##0"
                );
        }

        // =================================================
        // NGÀY SỬA
        // =================================================

        var currentDate =
            new Date();

        sheet.getRange(
            dongBatDauLuong + i,
            startColTho +
                listThoChinh.length +
                listThoPhu.length
        )
            .setValue(
                Utilities.formatDate(
                    currentDate,
                    Session.getScriptTimeZone(),
                    "HH:mm dd/MM/yyyy"
                )
            )
            .setFontSize(12);
    }

    // =====================================================
    // 10. SET O / P / Q
    // =====================================================

    columnDateL
        .setValues(dateL)
        .setFontSize(12);

    columnCustomerL
        .setValues(customerL)
        .setFontSize(12);

    columnBillL
        .setValues(billL)
        .setFontSize(12)
        .setNumberFormat("#,##0");

    // Giữ lại như logic cũ
    columnStatusDT.setValues(statusDT);

    // =====================================================
    // 11. TÍNH TỔNG LƯƠNG THỢ CHÍNH
    // =====================================================

    for (
        let i = 0;
        i < listThoChinh.length;
        i++
    ) {

        listThoChinh[i].tongLuong = 0;

        var listLuongChinh =
            sheet.getRange(
                dongBatDauLuong,
                listThoChinh[i].index,
                soDongChiTiet,
                1
            )
            .getValues();

        for (
            let l = 0;
            l < listLuongChinh.length;
            l++
        ) {

            listThoChinh[i].tongLuong +=
                Number(
                    listLuongChinh[l][0]
                ) || 0;
        }
    }

    // =====================================================
    // 12. TÍNH TỔNG LƯƠNG THỢ PHỤ
    // =====================================================

    for (
        let i = 0;
        i < listThoPhu.length;
        i++
    ) {

        listThoPhu[i].tongLuong = 0;

        var listLuongPhu =
            sheet.getRange(
                dongBatDauLuong,
                listThoPhu[i].index,
                soDongChiTiet,
                1
            )
            .getValues();

        for (
            let l = 0;
            l < listLuongPhu.length;
            l++
        ) {

            listThoPhu[i].tongLuong +=
                Number(
                    listLuongPhu[l][0]
                ) || 0;
        }
    }

    // =====================================================
    // 13. TỔNG DOANH THU
    // =====================================================

    var listDoanhThu =
        sheet.getRange(
            dongBatDauLuong,
            17, // Q
            soDongChiTiet,
            1
        )
        .getValues();

    tongDoanhThu = 0;

    for (
        let l = 0;
        l < listDoanhThu.length;
        l++
    ) {

        tongDoanhThu +=
            Number(
                listDoanhThu[l][0]
            ) || 0;
    }

    // =====================================================
    // 14. TỔNG LƯƠNG ĐƯA LÊN DÒNG 5-6
    // =====================================================

    // O5:P6
    sheet.getRange(
        dongTongLuong,
        15,
        2,
        2
    )
        .merge()
        .setValue("Tổng")
        .setFontSize(14)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // Q5:Q6
    sheet.getRange(
        dongTongLuong,
        17,
        2,
        1
    )
        .merge()
        .setValue(tongDoanhThu)
        .setFontSize(14)
        .setFontWeight("bold")
        .setNumberFormat("#,##0")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // R5:R6
    sheet.getRange(
        dongTongLuong,
        18,
        2,
        1
    )
        .merge()
        .setValue("Lương")
        .setFontSize(14)
        .setFontWeight("bold")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // =====================================================
    // TỔNG LƯƠNG THỢ CHÍNH
    // =====================================================

    for (
        let i = 0;
        i < listThoChinh.length;
        i++
    ) {

        sheet.getRange(
            dongTongLuong,
            listThoChinh[i].index,
            2,
            1
        )
            .merge()
            .setValue(
                listThoChinh[i].tongLuong
            )
            .setFontSize(14)
            .setFontWeight("bold")
            .setBackground(
                listThoChinh[i].color
            )
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    // =====================================================
    // TỔNG LƯƠNG THỢ PHỤ
    // =====================================================

    for (
        let i = 0;
        i < listThoPhu.length;
        i++
    ) {

        sheet.getRange(
            dongTongLuong,
            listThoPhu[i].index,
            2,
            1
        )
            .merge()
            .setValue(
                listThoPhu[i].tongLuong
            )
            .setFontSize(14)
            .setFontWeight("bold")
            .setBackground(
                listThoPhu[i].color
            )
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");
    }

    // =====================================================
    // CỘT NGÀY SỬA CỦA HÀNG TỔNG
    // =====================================================

    var cotNgaySua =
        startColTho +
        listThoChinh.length +
        listThoPhu.length;

    sheet.getRange(
        dongTongLuong,
        cotNgaySua,
        2,
        1
    )
        .merge()
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // =====================================================
    // 15. MERGE NGÀY + TỔNG BILL NGÀY
    // Chi tiết bắt đầu từ dòng 7
    // =====================================================

    var dataNgay =
        sheet.getRange(
            dongBatDauLuong,
            15, // O
            soDongChiTiet,
            1
        )
        .getValues();

    var groupStart = 0;

    var currentColor = 1;

    while (
        groupStart <
        dataNgay.length
    ) {

        var currentDateValue =
            dataNgay[groupStart][0];

        // Nếu dòng trống
        if (
            !currentDateValue
        ) {

            groupStart++;
            continue;
        }

        var currentDateString =
            Utilities.formatDate(
                currentDateValue,
                Session.getScriptTimeZone(),
                "dd/MM/yyyy"
            );

        var groupEnd =
            groupStart;

        // Tìm hết các dòng cùng ngày liên tiếp
        while (
            groupEnd + 1 <
                dataNgay.length &&
            dataNgay[groupEnd + 1][0]
        ) {

            var nextDateString =
                Utilities.formatDate(
                    dataNgay[
                        groupEnd + 1
                    ][0],
                    Session.getScriptTimeZone(),
                    "dd/MM/yyyy"
                );

            if (
                nextDateString !==
                currentDateString
            ) {
                break;
            }

            groupEnd++;
        }

        var groupLength =
            groupEnd -
            groupStart +
            1;

        var sheetStartRow =
            dongBatDauLuong +
            groupStart;

        // -----------------------------
        // Merge cột ngày O
        // -----------------------------

        var dateRange =
            sheet.getRange(
                sheetStartRow,
                15,
                groupLength,
                1
            );

        if (groupLength > 1) {

            var dateValue =
                dateRange
                    .getCell(1, 1)
                    .getValue();

            dateRange
                .clearContent()
                .merge()
                .setValue(dateValue);
        }

        dateRange
            .setFontWeight("bold")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");

        // -----------------------------
        // Tính tổng bill ngày
        // -----------------------------

        var billsInDay =
            sheet.getRange(
                sheetStartRow,
                17, // Q
                groupLength,
                1
            )
            .getValues();

        var totalDay = 0;

        for (
            var b = 0;
            b < billsInDay.length;
            b++
        ) {

            totalDay +=
                Number(
                    billsInDay[b][0]
                ) || 0;
        }

        // Cột R
        var totalDayRange =
            sheet.getRange(
                sheetStartRow,
                18,
                groupLength,
                1
            );

        if (groupLength > 1) {

            totalDayRange
                .clearContent()
                .merge()
                .setValue(totalDay);

        } else {

            totalDayRange
                .setValue(totalDay);
        }

        totalDayRange
            .setNumberFormat("#,##0")
            .setHorizontalAlignment("center")
            .setVerticalAlignment("middle");

        // -----------------------------
        // Màu xen kẽ theo ngày
        // Chỉ O:R
        // -----------------------------

        var hangColor;

        if (currentColor === 1) {

            hangColor =
                "#d9d2e9";

            currentColor = 2;

        } else {

            hangColor =
                "#ffffff";

            currentColor = 1;
        }

        sheet.getRange(
            sheetStartRow,
            15,
            groupLength,
            4
        )
            .setBackground(
                hangColor
            );

        // Sang nhóm ngày tiếp theo
        groupStart =
            groupEnd + 1;
    }

    // =====================================================
    // 16. BORDER TOÀN BỘ BẢNG
    // Dòng 5 -> hết chi tiết
    // =====================================================

    var soCotBangLuong =
        5 +
        listThoChinh.length +
        listThoPhu.length;

    var soDongBangLuong =
        dongCuoiChiTiet -
        dongTongLuong +
        1;

    sheet.getRange(
        dongTongLuong,
        cotBatDauLuong,
        soDongBangLuong,
        soCotBangLuong
    )
        .setBorder(
            true,
            true,
            true,
            true,
            true,
            true
        )
        .setHorizontalAlignment("center")
        .setVerticalAlignment("middle");

    // =====================================================
    // 17. XUẤT KPI TẠI J13:M
    // =====================================================

    xuatThongKeNhanVien(
        sheet,
        13,
        listThoChinh,
        listThoPhu
    );
}

        'portrait=true&' +
        'fitw=true&' +
        'sheetnames=false&' +
        'printtitle=false&' +
        'pagenumbers=true&' +
        'gridlines=false&' +
        'fzr=false&' +
        'gid=' + sheetId;

    var token = ScriptApp.getOAuthToken();
    var response = UrlFetchApp.fetch(url, {
        headers: {
            'Authorization': 'Bearer ' + token
        }
    });

    var pdfBlob = response.getBlob().setName(sheet.getName() + ".pdf");

    var email = "mrthanh260801@gmail.com";

        attachments: [pdfBlob]
    });

    Logger.log("Email đã được gửi với file PDF đính kèm.");
}

function filterMultipleColumns() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    sheet.getFilter().remove();

    var range = sheet.getRange("U4:S");
    var filter = range.createFilter();

    filter.setColumnFilterCriteria(
        21,
        SpreadsheetApp.newFilterCriteria()
            .whenNumberBetween(0, 500000000000000000)
            .build()
    );
}

    var blob = sheetToPDF(sheet, range);
    var file = folder.createFile(blob).setName("Exported_PDF.pdf");

    return file.getUrl();
}

function sheetToPDF(sheet, range) {
    var spreadsheet = sheet.getParent();
    var sheetId = sheet.getSheetId();
    var rangeA1 = range.getA1Notation();

    var url = `https://docs.google.com/spreadsheets/d/${spreadsheet.getId()}/export?`;
    url += `format=pdf&gid=${sheetId}&range=${rangeA1}`;
    url += "&portrait=true&size=A4";

    var params = {
        muteHttpExceptions: true,
        headers: {
            "Authorization": "Bearer " + ScriptApp.getOAuthToken()
        }
    };

    var response = UrlFetchApp.fetch(url, params);

    return response.getBlob().setName("Exported_PDF.pdf");
}

        .setWidth(400)

function exportRangeToPDF() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var spreadsheetId = SpreadsheetApp.getActiveSpreadsheet().getId();
    var sheetId = sheet.getSheetId();

    var startRow = 3;
    var startCol = 15; // O

    var lastRow = sheet.getLastRow() + 1;
    var lastCol = sheet.getLastColumn();

    var numRows = lastRow - startRow + 1;
    var numCols = lastCol - startCol + 1;

    var rangeA1 = sheet.getRange(
        startRow,
        startCol,
        numRows,
        numCols
    ).getA1Notation();

    var url =
        "https://docs.google.com/spreadsheets/d/" +
        spreadsheetId +
        "/export?format=pdf" +
        "&portrait=true" +
        "&size=A4" +
        "&gid=" +
        sheetId +
        "&range=" +
        rangeA1 +
        "&gridlines=false" +
        "&fitw=true";

    var token = ScriptApp.getOAuthToken();

    var response = UrlFetchApp.fetch(url, {
        headers: {
            "Authorization": "Bearer " + token
        }
    });

    var pdfBlob = response.getBlob()
        .setName("Bảng lương Thái Bình.pdf");

    var file = DriveApp.createFile(pdfBlob);
    var downloadUrl = file.getDownloadUrl();

    var html =
        '<script>window.open("' +
        downloadUrl +
        '", "_blank"); google.script.host.close();</script>';

    var ui = HtmlService.createHtmlOutput(html);

    SpreadsheetApp.getUi().showModalDialog(
        ui,
        "Đang tải xuống..."
    );
}
